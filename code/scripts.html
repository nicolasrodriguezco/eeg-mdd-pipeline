  raw_dir: data/raw

  processed_dir: data/processed
  reports_dir: reports
  manual_id_map: configs/manual_id_map.csv

dataset:
  # Extensiones que se consideran archivos de EEG.
  # MODMA ERP 128 canales (Lanzhou 2015): archivos .raw de EGI.

  # El ID son los 8 dígitos AL INICIO del nombre (ej. "02020019_erp 20150703 1052.raw").

  # que es la referencia de registro VREF/Cz y vale cero) a 250 Hz.
  expected_n_eeg_channels: 129
  expected_sfreq: 250
  allowed_flat_channels: [E129]

    # Excel oficial de MODMA (Sheet1: 53 filas; Sheet2 y Sheet3 vacías).
    # La etiqueta MDD/HC sale SIEMPRE de este archivo, nunca del prefijo del ID.

    id_column: "subject id"        # texto de 8 dígitos con cero inicial (02010002)
    group_column: type             # valores: MDD / HC
    # Cómo vienen escritos los grupos en el archivo clínico -> etiqueta final.
    group_map:
      MDD: MDD
      HC: HC

preprocessing:
  # Todo fijado en docs/preregistration.md (2026-11-04), antes de ver resultados.

  rename_channels: {E129: Cz}   # E129 = referencia de registro (VREF) = Cz del montaje
  l_freq: 0.5

  notch_freqs: [50.0]           # verificado: red de 50 Hz (China)
  reference: average
  resample_sfreq: 250           # los datos ya vienen a 250 Hz
  tmin: -0.2
  tmax: 0.6                     # el punto aparece >= 632 ms tras el rostro
  baseline: [-0.2, 0.0]
  reject_abs_uv: 100            # rechazo si |x| > 100 µV en cualquier canal tras la línea base
  reject_peak_to_peak_uv: null
  # Nombre del canal de EGI -> código fijo (igual en todos los sujetos).
  egi_event_codes: {fcue: 11, hcue: 12, scue: 13}
  event_id: {fear: 11, happy: 12, sad: 13}
  stim_channel: null

    highpass_for_fit: 1.0

    ensure_dirs(cfg)

    audit = audit_all(files, ds["subject_id_regex"], pp.get("stim_channel"))
    out = Path(cfg["paths"]["interim_dir"]) / "audit.csv"
    audit.to_csv(out, index=False)

    totals: Counter = Counter()
    for raw_counts in audit.get("event_counts", []).dropna():
        totals.update({int(k): v for k, v in json.loads(raw_counts).items()})
    print("\nCódigos de evento encontrados (código: total):")
    for code, n in sorted(totals.items()):
        print(f"  {code}: {n}")

    egi_totals: Counter = Counter()
    for raw_counts in audit.get("egi_event_counts", pd.Series(dtype=str)).dropna():
        egi_totals.update(json.loads(raw_counts))
    if egi_totals:
        print("\nEventos de EGI por nombre (nombre: total en todos los archivos):")
        for name, n in sorted(egi_totals.items()):
            print(f"  {name}: {n}")

    problems = check_audit(audit, ds.get("expected_n_eeg_channels"), ds.get("expected_sfreq"),
                           allowed_flat=ds.get("allowed_flat_channels", []))
    print(f"\nProblemas detectados: {len(problems)}")
    for p in problems:
        print(f"  - {p}")

"""Paso 3b: tabla demográfica por grupo y posibles confusores de registro.

Salidas:
  reports/tables/demographics.csv     edad, sexo, escolaridad (+ cuestionarios, solo descriptivos)
  reports/tables/recording_dates.csv  fecha de registro por sujeto (nombre de archivo y cabecera)
"""

import argparse
import json
from pathlib import Path

import _common  # noqa: F401
import pandas as pd

from eeg_mdd.config import load_config
from eeg_mdd.demographics import demographic_table, recording_datetime_from_name
from eeg_mdd.io import load_raw
from eeg_mdd.matching import load_clinical

DEMOGRAPHIC = {"age": "age", "education": "education（years）"}
QUESTIONNAIRES = ["PHQ-9", "GAD-7", "CTQ-SF", "LES", "SSRS", "PSQI"]   # SOLO descriptivos


def main() -> None:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--config", default=None)
    args = ap.parse_args()

    cfg = load_config(args.config)
    clin_cfg = cfg["dataset"]["clinical"]
    interim = Path(cfg["paths"]["interim_dir"])
    tables = Path(cfg["paths"]["reports_dir"]) / "tables"
    tables.mkdir(parents=True, exist_ok=True)

    subjects = pd.read_csv(interim / "subjects.csv", dtype={"eeg_id": str, "clinical_id": str})
    clinical = load_clinical(clin_cfg["file"], clin_cfg["id_column"], clin_cfg["group_column"], clin_cfg["group_map"])
    clinical = clinical.drop(columns="group")
    df = subjects.merge(clinical, on="clinical_id", how="left", validate="one_to_one")
    df = df.rename(columns={v: k for k, v in DEMOGRAPHIC.items()})

    demo = demographic_table(df, ["age", "education"] + QUESTIONNAIRES, ["gender"])
    demo["role"] = demo["variable"].map(lambda v: "questionnaire (descriptive only)" if v in QUESTIONNAIRES
                                        else "demographic (confounder analysis)")
    demo.to_csv(tables / "demographics.csv", index=False)
    cols = ["variable", "MDD_n", "MDD_mean_sd", "MDD_counts", "HC_n", "HC_mean_sd", "HC_counts", "test", "p_value"]
    print(demo[[c for c in cols if c in demo.columns]].to_string(index=False))

    # Fecha/hora de registro: el grupo podría estar confundido con la época de grabación.
    audit = pd.read_csv(interim / "audit.csv", dtype={"subject_id": str})
    rows = []

        raw = load_raw(f)

        rows.append({
            "subject_id": r["eeg_id"], "group": r["group"], "file": Path(f).name,
            "date_from_name": date.date() if date is not None else None, "time_from_name": hhmm,
            "meas_date_header": raw.info["meas_date"],
            "n_resp": counts.get("resp", 0),
            "extra_event_channels": ",".join(sorted(k for k in counts if k[:2] in ("DI", "D1", "D2", "Ca"))),
        })
    rec = pd.DataFrame(rows)
    rec.to_csv(tables / "recording_dates.csv", index=False)
    rec["month"] = pd.to_datetime(rec["date_from_name"]).dt.to_period("M")
    print("\nSujetos por mes de registro y grupo (fecha del nombre de archivo):")
    print(pd.crosstab(rec["month"], rec["group"]).to_string())
    print(f"\nTablas en {tables}")

04_preprocess.py: preprocessing and exclusion
"""Paso 4: preprocesamiento por sujeto (en paralelo) y aplicación de los criterios de exclusión.

Salidas:
  data/processed/epochs/sub-<id>-epo.fif        épocas -200..600 ms alineadas al par de rostros
  data/processed/continuous/sub-<id>-seg-epo.fif segmentos limpios de 2 s a 125 Hz
  data/processed/ica/sub-<id>-ica.fif
  configs/ica_exclusions.csv                     componentes quitados (ICLabel-auto)
  reports/preprocessing_log.csv                  qué pasó con cada sujeto
  reports/ica_labels.csv                         etiqueta ICLabel de cada componente
  reports/qc/sub-<id>.html                       mne.Report por sujeto
  data/interim/subjects_final.csv                sujetos que cumplen los criterios
"""

import argparse
import traceback
from pathlib import Path

import _common  # noqa: F401
import pandas as pd
from joblib import Parallel, delayed

from eeg_mdd.checks import check_labels, raise_if_problems
from eeg_mdd.config import ensure_dirs, load_config
from eeg_mdd.io import load_raw
from eeg_mdd.preprocess import apply_exclusion_criteria, load_ica_exclusions, preprocess_subject


def _make_report(res, sid: str, group: str, out: Path) -> None:
    import matplotlib

    matplotlib.use("Agg")
    import mne

    rep = mne.Report(title=f"sub-{sid} ({group})", verbose="error")
    if res.ica is not None:
        rep.add_ica(res.ica, title="ICA (excluded components marked)", inst=None, n_jobs=1)
        if res.ica_labels is not None:
            rep.add_html(res.ica_labels.round(3).to_html(index=False), title="ICLabel")
    rep.add_html(pd.Series(res.bad_reasons, dtype=str).to_frame("reason").to_html(), title="Bad channels")
    rep.add_epochs(res.epochs, title="Epochs (face-locked)", psd=False)
    evokeds = [res.epochs[c].average() for c in res.epochs.event_id if len(res.epochs[c])]
    rep.add_evokeds(evokeds, n_time_points=5)
    rep.save(out, overwrite=True, open_browser=False, verbose="error")


def run_one(sid: str, group: str, file: str, pp: dict, manual_excl, dirs: dict, report: bool) -> dict:
    try:
        raw = load_raw(file)
        res = preprocess_subject(raw, pp, sid, manual_excl, ica_dir=dirs["ica"])
        res.epochs.save(dirs["epochs"] / f"sub-{sid}-epo.fif", overwrite=True, verbose="error")
        if res.continuous is not None:
            res.continuous.save(dirs["cont"] / f"sub-{sid}-seg-epo.fif", overwrite=True, verbose="error")
        if res.ica_labels is not None:
            res.ica_labels.assign(subject_id=sid).to_csv(dirs["labels"] / f"sub-{sid}.csv", index=False)
        if report:
            _make_report(res, sid, group, dirs["qc"] / f"sub-{sid}.html")
        log = res.log
        log["error"] = ""
    except Exception as exc:  # noqa: BLE001 - se registra y se sigue con los demás
        log = {"subject_id": sid, "error": f"{type(exc).__name__}: {exc}",
               "traceback": traceback.format_exc(limit=3)}
    log["group"] = group
    return log


def main() -> None:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--config", default=None)
    ap.add_argument("--subject", default=None, help="Procesar solo este sujeto")
    ap.add_argument("--n-jobs", type=int, default=1)
    ap.add_argument("--no-report", action="store_true")
    args = ap.parse_args()

    cfg = load_config(args.config)
    ensure_dirs(cfg)
    paths, pp = cfg["paths"], cfg["preprocessing"]
    interim, processed, reports = Path(paths["interim_dir"]), Path(paths["processed_dir"]), Path(paths["reports_dir"])
    dirs = {"epochs": processed / "epochs", "cont": processed / "continuous", "ica": processed / "ica",
            "labels": interim / "ica_labels", "qc": reports / "qc"}
    for d in dirs.values():
        d.mkdir(parents=True, exist_ok=True)

    subjects = pd.read_csv(interim / "subjects.csv", dtype={"eeg_id": str})
    raise_if_problems(check_labels(subjects), "etiquetas")
    audit = pd.read_csv(interim / "audit.csv", dtype={"subject_id": str})
    file_by_id = dict(zip(audit["subject_id"], audit["file"]))
    manual = load_ica_exclusions(paths["ica_exclusions"])
    if args.subject:
        subjects = subjects[subjects["eeg_id"] == args.subject]

    logs = Parallel(n_jobs=args.n_jobs, verbose=5)(
        delayed(run_one)(s["eeg_id"], s["group"], file_by_id[s["eeg_id"]], pp, manual.get(s["eeg_id"]), dirs,
                         not args.no_report)
        for _, s in subjects.iterrows()
    )
    log = pd.DataFrame(logs)
    conditions = list(pp["event_id"])
    n_ch = int(cfg["dataset"].get("expected_n_eeg_channels") or 129)
    crit = cfg.get("exclusion") or {"max_bad_channel_frac": 1.0, "min_trials_total": 0, "min_trials_per_condition": 0}
    log = apply_exclusion_criteria(log, crit, n_ch, conditions)
    out = reports / ("preprocessing_log.csv" if not args.subject else f"preprocessing_log_{args.subject}.csv")
    log.to_csv(out, index=False)

    if not args.subject:
        labels = [pd.read_csv(p, dtype={"subject_id": str}) for p in sorted(dirs["labels"].glob("sub-*.csv"))]
        if labels:
            lab = pd.concat(labels)
            lab.to_csv(reports / "ica_labels.csv", index=False)
            excl = (lab[lab["exclude"]].groupby("subject_id")["component"]
                    .agg(lambda s: ",".join(str(int(c)) for c in s)))
            rows = pd.DataFrame({"subject_id": log["subject_id"]})
            rows["exclude_components"] = rows["subject_id"].map(excl).fillna("")
            rows["reviewer"] = f"ICLabel-auto (p>={pp['ica']['iclabel_threshold']})"
            rows["notes"] = "automatic, rule fixed in docs/preregistration.md"
            rows.to_csv(paths["ica_exclusions"], index=False)
        final = subjects.merge(log[["subject_id", "included", "exclusion_reason"]],
                               left_on="eeg_id", right_on="subject_id", how="left")
        final[final["included"]].drop(columns=["subject_id"]).to_csv(interim / "subjects_final.csv", index=False)
        print(f"\nIncluidos: {int(final['included'].sum())}/{len(final)}")
        print(final.loc[~final["included"], ["eeg_id", "group", "exclusion_reason"]].to_string(index=False))
    print(f"Registro: {out}")
    print(log.drop(columns=[c for c in log.columns if c in ("traceback",)]).describe().T.round(1).to_string())


if __name__ == "__main__":
    main()

06_features.py: subject-level features and trial-level data
"""Paso 6: características por sujeto (biomarcadores) y datos por ensayo para Riemann/DL.

Las características no usan etiquetas, así que pueden calcularse una vez por sujeto;
todo lo que aprende de los datos (escalado, selección, modelos) va dentro de la CV.

Salidas (data/processed/features/):
  subject_features.parquet   una fila por sujeto: subject_id, group + ~390 características
  channel_power.parquet      potencia absoluta por canal y banda (para topomapas)
  trial_regions.npz          señal regional por ensayo (n_trials, 10, 201) + subject_id + condición
  evokeds/sub-<id>-ave.fif   ERP promedio por condición
"""

import argparse
from pathlib import Path

import _common  # noqa: F401
import mne
import numpy as np
import pandas as pd
from joblib import Parallel, delayed

from eeg_mdd.config import load_config
from eeg_mdd.features import subject_features
from eeg_mdd.features.spectral import BANDS


def one(sid: str, group: str, processed: Path, out: Path) -> dict:
    ep = mne.read_epochs(processed / "epochs" / f"sub-{sid}-epo.fif", verbose="error")
    seg = mne.read_epochs(processed / "continuous" / f"sub-{sid}-seg-epo.fif", verbose="error")
    res = subject_features(ep, seg)
    mne.write_evokeds(out / "evokeds" / f"sub-{sid}-ave.fif", res["evokeds"], overwrite=True, verbose="error")
    row = {"subject_id": sid, "group": group, **res["features"]}

def run(name, est, grid, X, y, g, out_dir, n_perm, n_jobs, extra=None, kind="tabular"):
    if RESUME and (out_dir / name / "meta.json").exists() and (out_dir / name / "ci.csv").exists():
        print(f"  {name:<28} ya calculado -> se omite (--no-resume para recalcular)", flush=True)
        return None
    # Procesos según la RAM libre en este momento (nunca más que n_jobs).
    n_jobs = safe_n_jobs(PER_JOB_GB[kind], max_jobs=n_jobs)
    print(f"  {name}: {n_jobs} procesos (RAM libre {free_gb():.1f} GB)", flush=True)
    t0 = time.time()
    res = nested_cv(name, est, grid, X, y, g, N_SPLITS, N_REPEATS, N_INNER, n_jobs=n_jobs)

        perm = permutation_test(est, grid, X, y, g, obs, n_perm=n_perm, n_splits=N_SPLITS, n_inner=N_INNER,

            **(extra or {})}
    return save(res, out_dir / name, perm, meta)


def trialwise_control(X, y, g, out: Path) -> pd.DataFrame:
    """INVÁLIDO A PROPÓSITO: separación por ensayo (como la versión vieja), solo para medir la inflación."""
    rows = []
    models = {"TS-LR": riemann_models()["TS-LR"][0].set_params(clf__C=0.1),
              "RawAmplitude-RF (old approach)": None}
    Xflat = np.asarray(X).reshape(len(X), -1)[:, ::2].astype(np.float32)                 # señal regional cruda, 1 de cada 2 muestras
    for name, est in models.items():
        for split in ("trial-wise (INVALID)", "subject-wise"):
            for r in range(3):
                if split.startswith("trial"):
                    folds = StratifiedKFold(5, shuffle=True, random_state=r).split(X, y)
                else:
                    from eeg_mdd.evaluation.cv import outer_splits
                    folds = [(tr, te) for _, _, tr, te in outer_splits(y, g, 5, 1, seed0=r)]
                prob = np.zeros(len(y))
                for tr, te in folds:
                    if est is None:
                        m = RandomForestClassifier(n_estimators=300, max_features="sqrt", min_samples_leaf=3,
                                                   class_weight="balanced", n_jobs=safe_n_jobs(0.5),
                                                   random_state=r)
                        m.fit(Xflat[tr], y[tr])
                        prob[te] = m.predict_proba(Xflat[te])[:, 1]
                    else:
                        from sklearn.base import clone
                        m = clone(est).fit(np.asarray(X[tr]), y[tr])
                        prob[te] = m.predict_proba(np.asarray(X[te]))[:, 1]
                trial_auc = roc_auc_score(y, prob)
                subj = pd.DataFrame({"g": g, "y": y, "p": prob}).groupby("g").agg(y=("y", "first"), p=("p", "mean"))
                rows.append({"model": name, "split": split, "repeat": r, "trial_auc": trial_auc,
                             "subject_auc": roc_auc_score(subj["y"], subj["p"])})
                print(f"  control {name:<32} {split:<22} rep{r} trial AUC={trial_auc:.3f} "
                      f"subject AUC={rows[-1]['subject_auc']:.3f}", flush=True)
    df = pd.DataFrame(rows)
    out.mkdir(parents=True, exist_ok=True)
    df.to_csv(out / "inflation_control.csv", index=False)
    return df

def load_trials(processed: Path, subjects: pd.DataFrame):
    """Épocas de todos los sujetos en un memmap float32 en disco (no ocupa RAM entera)."""
    from eeg_mdd.resources import as_memmap

    cache = processed / "features" / "trials_full_X.npy"
    meta = processed / "features" / "trials_full_meta.npz"
    if cache.exists() and meta.exists():
        m = np.load(meta, allow_pickle=True)
        if list(m["subjects"]) == list(subjects["eeg_id"]):
            return np.load(cache, mmap_mode="r"), m["y"], m["g"], float(m["sfreq"])
    X, y, g = [], [], []
    for _, s in subjects.iterrows():
        ep = mne.read_epochs(processed / "epochs" / f"sub-{s['eeg_id']}-epo.fif", verbose="error")
        d = (ep.get_data(picks="eeg") * 1e6).astype(np.float32)          # µV
        X.append(d)

    del X

    del Xa
    np.savez(meta, y=y, g=g, sfreq=sf, subjects=np.array(subjects["eeg_id"]))
    return mm, y, g, sf


def run_model(name, X, y, g, sfreq, n_repeats, cfg, device, perm_seed=None):
    yy = y.copy()
    if perm_seed is not None:                       # etiquetas permutadas ENTRE sujetos
        subj = pd.DataFrame({"g": g, "y": y}).drop_duplicates("g")
        rng = np.random.default_rng(10_000 + perm_seed)
        mp = dict(zip(subj["g"], rng.permutation(subj["y"].to_numpy())))
        yy = np.array([mp[s] for s in g])
    oof, folds, info = [], [], []
    for r, k, tr, te in outer_splits(yy, g, 5, n_repeats):
        cfg_k = DLConfig(**{**cfg.__dict__, "seed": 1000 * r + k})
        out = train_fold(name, X, yy, g, tr, te, sfreq, cfg_k, device)
        subj = aggregate_to_subject(out["prob"], yy[te], g[te]).assign(repeat=r, fold=k)
        oof.append(subj)
        folds.append({"repeat": r, "fold": k, **subject_metrics(subj["y"], subj["prob"])})
        info.append({"repeat": r, "fold": k, "epochs": out["epochs"], "best_val_auc": out["best_val_auc"],
                     "seconds": round(out["seconds"], 1)})
        if perm_seed is None:
            print(f"  {name} rep{r} fold{k}: test AUC={folds[-1]['auc']:.3f} val AUC={out['best_val_auc']:.3f} "
                  f"epochs={out['epochs']} {out['seconds']:.0f}s", flush=True)
    oof = pd.concat(oof, ignore_index=True)
    rep = pd.DataFrame([{"repeat": r, **subject_metrics(d["y"], d["prob"])} for r, d in oof.groupby("repeat")])
    return CVResult(name, oof, pd.DataFrame(folds), rep, info)

def cohen_d(a, b):
    a, b = np.asarray(a, float), np.asarray(b, float)
    sp = np.sqrt(((len(a) - 1) * a.var(ddof=1) + (len(b) - 1) * b.var(ddof=1)) / (len(a) + len(b) - 2))
    return (a.mean() - b.mean()) / sp


def _fold_importance(pipe, grid, X, y, g, tr, te, fams, n_shuffle, seed):
    inner = list(StratifiedGroupKFold(5, shuffle=True, random_state=seed).split(np.zeros(len(tr)), y[tr], g[tr]))
    gs = GridSearchCV(clone(pipe), grid, scoring="roc_auc", cv=inner, n_jobs=1).fit(X.iloc[tr], y[tr])
    m = gs.best_estimator_
    Xte = X.iloc[te].copy()
    base = roc_auc_score(y[te], m.predict_proba(Xte)[:, 1])
    rng = np.random.default_rng(seed)
    feat_imp, fam_imp = {}, {}
    for c in X.columns:
        drops = []
        for _ in range(n_shuffle):
            Xp = Xte.copy()
            Xp[c] = rng.permutation(Xp[c].to_numpy())
            drops.append(base - roc_auc_score(y[te], m.predict_proba(Xp)[:, 1]))
        feat_imp[c] = float(np.mean(drops))
    for fam, cols in fams.items():
        drops = []
        for _ in range(n_shuffle):
            Xp = Xte.copy()
            perm = rng.permutation(len(Xp))
            Xp[cols] = Xp[cols].to_numpy()[perm]          # permuta la familia entera junta
            drops.append(base - roc_auc_score(y[te], m.predict_proba(Xp)[:, 1]))
        fam_imp[fam] = float(np.mean(drops))
    coef = pd.Series(m.named_steps["clf"].coef_[0], index=X.columns)
    return feat_imp, fam_imp, coef

        name, cat = DISPLAY.get(d.name, (d.name, "Other"))
        row = {"key": d.name, "model": name, "category": cat, "n_repeats": rep["repeat"].nunique(),
               "p_perm": meta.get("p_value", np.nan), "n_perm": meta.get("n_perm", 0),
               "null_q95": meta.get("null_q95", np.nan), "auc_sd_repeats": rep["auc"].std()}
        for m in METRICS:
            row[m] = ci.loc[m, "estimate"]
            row[f"{m}_lo"] = ci.loc[m, "ci_low"]
            row[f"{m}_hi"] = ci.loc[m, "ci_high"]
        rows.append(row)
    df = pd.DataFrame(rows)
    df["cat_rank"] = df["category"].map({c: i for i, c in enumerate(CATEGORY_ORDER)}).fillna(99)
    return df.sort_values(["cat_rank", "auc"], ascending=[True, False]).reset_index(drop=True)


def fmt_ci(v, lo, hi, nd=2):
    return f"{v:.{nd}f} [{lo:.{nd}f}, {hi:.{nd}f}]"


def fmt_p(p, n):
    if not np.isfinite(p) or not n:
        return "--"
    floor = 1 / (n + 1)
    return f"$<${floor:.3f}" if p <= floor + 1e-12 else f"{p:.3f}"


def latex_main_table(df: pd.DataFrame, path: Path) -> None:
    lines = [r"\begin{tabular}{@{}llllllr@{}}", r"\toprule",
             r"Model & AUC [95\% CI] & Bal.\ acc.\ [95\% CI] & Sens. & Spec. & Brier & $p_\text{perm}$ \\",
             r"\midrule"]
    for cat in CATEGORY_ORDER:
        sub = df[df["category"] == cat]
        if sub.empty:
            continue
        lines.append(rf"\multicolumn{{7}}{{@{{}}l}}{{\textit{{{cat}}}}} \\")
        for _, r in sub.iterrows():
            name = r["model"].replace("&", r"\&")
            if "primary" in name:
                name = r"\textbf{" + name + "}"
            perm = fmt_p(r["p_perm"], r["n_perm"]) + (f" ({int(r['n_perm'])})" if r["n_perm"] else "")
            lines.append(f"\\quad {name} & {fmt_ci(r['auc'], r['auc_lo'], r['auc_hi'])} & "
                         f"{fmt_ci(r['balanced_accuracy'], r['balanced_accuracy_lo'], r['balanced_accuracy_hi'])} & "
                         f"{r['sensitivity']:.2f} & {r['specificity']:.2f} & {r['brier']:.3f} & {perm} \\\\")
    lines += [r"\bottomrule", r"\end{tabular}"]
    path.write_text("\n".join(lines), encoding="utf-8")

def fig_roc(models_dir: Path, keys: list[str], path: Path) -> None:

def _comparisons(models_dir: Path, df: pd.DataFrame, macros: dict, key) -> list[str]:
    """ΔAUC por pliegue frente al primario y a la línea base demográfica (t corregido) + Holm sobre secundarios."""
    from statsmodels.stats.multitest import multipletests

    from eeg_mdd.evaluation.nested import corrected_ttest

    def folds(k):
        f = models_dir / k / "fold_scores.csv"
        return pd.read_csv(f).set_index(["repeat", "fold"])["auc"] if f.exists() else None

    ref = {"primary": folds("PRIMARY_LR-L2_all"), "demo": folds("Demographics-LR")}
    if any(v is None for v in ref.values()):
        return []
    n_test, n_train = 9, 36                     # 45 participantes, 5 pliegues externos
    sec = df[~df["key"].isin(["PRIMARY_LR-L2_all", "Demographics-LR"]) & (df["n_perm"] > 0)].copy()
    sec["p_holm"] = multipletests(sec["p_perm"].to_numpy(), method="holm")[1] if len(sec) else []
    holm = dict(zip(sec["key"], sec["p_holm"]))
    rows = [r"\begin{tabular}{@{}lrrrrrr@{}}", r"\toprule",
            r" & \multicolumn{2}{c}{vs.\ primary} & \multicolumn{2}{c}{vs.\ demographics} & & \\",
            r"\cmidrule(lr){2-3}\cmidrule(lr){4-5}",
            r"Model & $\Delta$AUC & $p_\text{corr}$ & $\Delta$AUC & $p_\text{corr}$ & $p_\text{perm}$ & $p_\text{Holm}$ \\",
            r"\midrule"]
    for _, r in df[df["key"] != "Demographics-LR"].iterrows():
        a = folds(r["key"])
        cells = []
        for nm, b in ref.items():
            if r["key"] == "PRIMARY_LR-L2_all" and nm == "primary":
                cells += ["--", "--"]
                continue
            j = pd.concat([a, b], axis=1, join="inner").dropna()
            t = corrected_ttest((j.iloc[:, 0] - j.iloc[:, 1]).to_numpy(), n_train, n_test)
            cells += [f"{t['mean_diff']:+.2f}", f"{t['p_value']:.3f}"]
            macros[f"Cmp{key(r['key'])}Vs{nm.title()}Diff"] = f"{t['mean_diff']:+.2f}"
            macros[f"Cmp{key(r['key'])}Vs{nm.title()}P"] = f"{t['p_value']:.3f}"
        ph = holm.get(r["key"], np.nan)
        if np.isfinite(ph):
            macros[f"{key(r['key'])}Pholm"] = f"{ph:.3f}"
        name = r["model"].replace("&", r"\&")
        rows.append(f"{name} & " + " & ".join(cells) + f" & {fmt_p(r['p_perm'], r['n_perm'])} & "
                    f"{(f'{ph:.3f}' if np.isfinite(ph) else '--')} \\\\")
    macros["NSecondaryHolm"] = str(len(sec))
    rows += [r"\bottomrule", r"\end{tabular}"]
    return rows

def _region_mean(values: np.ndarray, ch_names: list[str], chans: list[str]) -> float:
    idx = [ch_names.index(c) for c in chans if c in ch_names]

def one_subject(sid: str, raw_file: str, bads: list[str], processed: Path, pp: dict) -> dict:
    row: dict = {"subject_id": sid}
    # --- crudo: conducta y músculo
    raw = prepare_raw(load_raw(raw_file, preload=True), pp)
    sf = raw.info["sfreq"]
    dots = {c: event_onsets(raw, ch) / sf for c, ch in CONDS.items() if ch in raw.ch_names}
    resp = event_onsets(raw, "resp") / sf if "resp" in raw.ch_names else np.array([])
    row.update(behavioural_rt(dots, resp))
    starts = [event_onsets(raw, n) for n in ("fcue", "hcue", "scue") if n in raw.ch_names]
    t0 = max(0.0, min(s.min() for s in starts) / sf - 1.0)
    t1 = min(raw.times[-1], max(s.max() for s in starts) / sf + 2.0)
    eeg = raw.copy().pick("eeg").crop(t0, t1)
    eeg.info["bads"] = [b for b in bads if b in eeg.ch_names]
    eeg.drop_channels(eeg.info["bads"])
    eeg.filter(1.0, None, verbose="error").notch_filter([50.0, 100.0], verbose="error")
    eeg.set_eeg_reference("average", verbose="error")
    hf, hf_rel = muscle_index(eeg.get_data() * 1e6, sf)
    chs = eeg.ch_names
    row["emg_logpow_global"] = float(np.mean(hf))
    row["emg_rel_global"] = float(np.mean(hf_rel))
    for reg, chans in REGIONS.items():
        row[f"emg_rel_{reg}"] = _region_mean(hf_rel, chs, chans)
    del raw, eeg

    # --- segmentos limpios: aperiódico, pico alfa y theta de línea media
    seg = mne.read_epochs(processed / "continuous" / f"sub-{sid}-seg-epo.fif", verbose="error")
    d = seg.get_data(picks="eeg")
    sfs = seg.info["sfreq"]
    f, p = welch(d, fs=sfs, nperseg=d.shape[-1], axis=-1)
    psd = p.mean(0) * 1e12                         # (canales, frecuencias) en µV²/Hz
    ch = seg.ch_names
    used = sorted({c for v in REGIONS.values() for c in v})
    groups = {**REGIONS, "global": used}
    for name, chans in groups.items():
        idx = [ch.index(c) for c in chans if c in ch]
        off, exp_ = aperiodic_fit(f, psd[idx].mean(0))
        row[f"ap_exponent_{name}"], row[f"ap_offset_{name}"] = exp_, off
    idx = [ch.index(c) for c in POSTERIOR if c in ch]
    pp_ = psd[idx].mean(0)
    off, exp_ = aperiodic_fit(f, pp_)
    row["alpha_peak_freq_post"], row["alpha_peak_power_post"] = alpha_peak(f, aperiodic_residual(f, pp_, off, exp_))
    idx = [ch.index(c) for c in FM_THETA if c in ch]
    fm = psd[idx].mean(0)
    th = (f >= 4) & (f < 8)
    tot = (f >= 1) & (f < 40)
    off, exp_ = aperiodic_fit(f, fm)
    row["fmtheta_logpow"] = float(np.log10(np.trapezoid(fm[th], f[th])))
    row["fmtheta_rel"] = float(np.trapezoid(fm[th], f[th]) / np.trapezoid(fm[tot], f[tot]))
    row["fmtheta_periodic"] = float(np.mean(aperiodic_residual(f, fm, off, exp_)[th]))
    del seg, d

    # --- variabilidad P300 de ensayo a ensayo
    ep = mne.read_epochs(processed / "epochs" / f"sub-{sid}-epo.fif", verbose="error")
    cp = [c for c in erp_clusters()["centroparietal"] if c in ep.ch_names]
    x = ep.get_data(picks=cp).mean(1) * 1e6
    w = (ep.times >= 0.25) & (ep.times <= 0.5)
    amp = x[:, w].mean(1)
    row["p300_trial_sd"] = float(np.std(amp, ddof=1))
    row["p300_trial_snr"] = float(np.mean(amp) / np.std(amp, ddof=1))
    return row


def derived(feats: pd.DataFrame) -> pd.DataFrame:

def block_permutation(est, grid, X: np.ndarray, y, g, eeg_idx: np.ndarray, observed: float, n_perm: int,
                      n_jobs: int) -> dict:
    """Nula de "el EEG no añade nada": se barajan entre sujetos las FILAS del bloque de EEG.

    Las etiquetas y la demografía se dejan fijas, así que se conserva la relación demografía–diagnóstico y solo se
    rompe la del EEG con ambas.
    """
    def one(i):
        rng = np.random.default_rng(20_000 + i)
        Xp = X.copy()
        Xp[:, eeg_idx] = X[rng.permutation(len(X))][:, eeg_idx]
        r = nested_cv("perm", est, grid, Xp, y, g, N_SPLITS, 1, N_INNER, n_jobs=1)
        return float(r.repeat_scores["auc"].iloc[0])

    null = np.array(Parallel(n_jobs=n_jobs)(delayed(one)(i) for i in range(n_perm)))
    return {"observed_auc": observed, "n_perm": n_perm, "p_value": float((1 + (null >= observed).sum()) / (1 + n_perm)),
            "null_mean": float(null.mean()), "null_q95": float(np.quantile(null, 0.95)), "null": null}

def part_models(cfg, allf, compact, feat_dir, out, n_jobs, n_perm):
    inc = allf[allf["included"]].reset_index(drop=True)
    y = (inc["group"] == "MDD").astype(int).to_numpy()
    g = inc["subject_id"].to_numpy()
    cols = [c for c in allf.columns if family_of(c)]
    beh = ["beh_rt_median", "beh_rt_sd", "beh_rt_cv", "beh_miss_rate", "beh_rt_median_fear", "beh_rt_median_happy",
           "beh_rt_median_sad", "beh_rt_sad_minus_happy"]
    lr = {"clf__C": C_GRID}
    rows = []

    def done(name):
        f = out / name / "row.json"
        if f.exists():
            rows.append(json.loads(f.read_text()))
            print(f"  {name}: ya calculado", flush=True)
            return True
        return False

    def std(name, pipe, grid, X, extra):
        if done(name):
            return
        t0 = time.time()
        res = nested_cv(name, pipe, grid, X, y, g, N_SPLITS, N_REPEATS, N_INNER, n_jobs=n_jobs)
        perm = permutation_test(pipe, grid, X, y, g, float(res.repeat_scores["auc"].iloc[0]), n_perm=n_perm,
                                n_jobs=n_jobs)
        rows.append(save(res, out / name, perm, {**extra, "seconds": round(time.time() - t0, 1)}))

    # 1. Valor incremental: demografía + EEG, con la nula que baraja el bloque de EEG
    for name, ec in (("Demo+EEG288", cols), ("Demo+Compact", compact)):
        if done(name):
            continue
        X = inc[ec + DEMO].to_numpy(float)
        pipe = _tab(_lr("l2"))
        res = nested_cv(name, pipe, lr, X, y, g, N_SPLITS, N_REPEATS, N_INNER, n_jobs=n_jobs)
        perm = block_permutation(pipe, lr, X, y, g, np.arange(len(ec)), float(res.repeat_scores["auc"].iloc[0]),
                                 n_perm, n_jobs)
        rows.append(save(res, out / name, perm, {"n_features": len(ec) + 3,
                                                 "test": "EEG-block permutation (EEG adds nothing)"}))
    # 2. EEG desconfundido de edad, sexo y escolaridad dentro del pliegue
    deconf = Pipeline([("impute", SimpleImputer(strategy="median")), ("deconf", ConfoundRegressor(3)),
                       ("scale", StandardScaler()), ("clf", _lr("l2"))])
    std("EEG288-deconfounded", deconf, lr, inc[cols + DEMO].to_numpy(float),
        {"n_features": len(cols), "test": "label permutation"})
    # 3. PCA dentro del pliegue
    pca = Pipeline([("impute", SimpleImputer(strategy="median")), ("scale", StandardScaler()),
                    ("pca", PCA(random_state=0)), ("clf", _lr("l2"))])
    std("EEG288-PCA-LR", pca, {"pca__n_components": [5, 10, 20], "clf__C": C_GRID}, inc[cols].to_numpy(float),
        {"n_features": len(cols), "test": "label permutation"})
    # 4. Conjunto compacto con base teórica
    std("Compact-LR", _tab(_lr("l2")), lr, inc[compact].to_numpy(float),
        {"n_features": len(compact), "test": "label permutation"})
    # 5. Conducta (TR): sin EEG ni cuestionarios
    std("Behaviour-RT-LR", _tab(_lr("l2")), lr, inc[beh].to_numpy(float),
        {"n_features": len(beh), "test": "label permutation"})
    # 6. Riemann por sujeto: covarianza media por sujeto (sin etiquetas) → espacio tangente dentro del pliegue
    from pyriemann.tangentspace import TangentSpace
    from pyriemann.utils.mean import mean_covariance

    std("Subject-Riemann-TS-LR", riem, lr, S, {"n_features": 55, "test": "label permutation"})

    tab = pd.DataFrame(rows)
    from statsmodels.stats.multitest import multipletests
    tab["p_holm"] = multipletests(tab["p_perm"], method="holm")[1]
    # Comparación con la demografía y con el primario en los mismos pliegues (Nadeau–Bengio)
    mres = Path(cfg["paths"]["reports_dir"]) / "results" / "models"
    ref = {k: pd.read_csv(mres / k / "fold_scores.csv").set_index(["repeat", "fold"])["auc"]
           for k in ("Demographics-LR", "PRIMARY_LR-L2_all")}
    for i, r in tab.iterrows():

    print(tab.round(3).to_string())

def part_power(cfg, allf, out, n_jobs, n_sims):
    rng = np.random.default_rng(0)
    null = np.load(Path(cfg["paths"]["reports_dir"]) / "results" / "models" / "PRIMARY_LR-L2_all" / "perm_null.npy")
    thr = float(np.quantile(null, 0.95))
    n1, n0 = 18, 27
    # (a) Puntuación ideal binormal: el mejor caso, sin costo de aprendizaje
    ideal = []
    for auc in (0.6, 0.65, 0.7, 0.75, 0.8, 0.85, 0.9):
        d = np.sqrt(2) * stats.norm.ppf(auc)
        hits = 0
        for _ in range(20000):
            a, b = rng.normal(d, 1, n1), rng.normal(0, 1, n0)
            hits += (a[:, None] > b[None, :]).mean() >= thr
        ideal.append({"true_auc": auc, "power_ideal_score": hits / 20000})
    # (b) Semisintético: las 288 variables reales con las filas barajadas (sin señal real) más un desplazamiento de
    #     d DE en k = 10 variables al azar; se corre el pipeline primario completo (1 repetición).
    inc = allf[allf["included"]].reset_index(drop=True)
    cols = [c for c in allf.columns if family_of(c)]
    X0 = inc[cols].to_numpy(float)
    y = (inc["group"] == "MDD").astype(int).to_numpy()
    g = inc["subject_id"].to_numpy()
    pipe, grid = _tab(_lr("l2")), {"clf__C": C_GRID}

    def sim(d, i):
        r = np.random.default_rng(1000 * int(d * 100) + i)
        X = X0[r.permutation(len(X0))].copy()
        S = r.choice(X.shape[1], 10, replace=False)
        sd = X[:, S].std(0)
        X[:, S] += d * sd * y[:, None]
        z = (X[:, S] - X[:, S].mean(0)) / sd
        from sklearn.metrics import roc_auc_score
        oracle = roc_auc_score(y, z.sum(1))
        res = nested_cv("sim", pipe, grid, X, y, g, N_SPLITS, 1, N_INNER, seed0=i, n_jobs=1)
        return d, oracle, float(res.repeat_scores["auc"].iloc[0])

    t0 = time.time()
    sims = Parallel(n_jobs=n_jobs)(delayed(sim)(d, i) for d in (0.0, 0.3, 0.5, 0.8, 1.0) for i in range(n_sims))
    s = pd.DataFrame(sims, columns=["d_per_feature", "oracle_auc", "cv_auc"])
    semi = s.groupby("d_per_feature").agg(oracle_auc=("oracle_auc", "mean"), cv_auc_mean=("cv_auc", "mean"),
                                          power=("cv_auc", lambda v: float((v >= thr).mean()))).reset_index()
    pd.DataFrame(ideal).to_csv(out / "power_ideal.csv", index=False)
    semi.to_csv(out / "power_semisynthetic.csv", index=False)
    (out / "power_meta.json").write_text(json.dumps({"threshold_null_q95": thr, "n_sims": n_sims, "k": 10,
                                                     "seconds": round(time.time() - t0, 1)}))
    print(f"umbral {thr:.3f}\n", pd.DataFrame(ideal).to_string(), "\n", semi.round(3).to_string())

def part_curve(cfg, allf, out, n_jobs):
    inc = allf[allf["included"]].reset_index(drop=True)
    cols = [c for c in allf.columns if family_of(c)]
    y = (inc["group"] == "MDD").astype(int).to_numpy()
    g = inc["subject_id"].to_numpy()
    sets = {"Primary LR-L2 (EEG 288)": inc[cols].to_numpy(float), "Demographics LR": inc[DEMO].to_numpy(float)}
    pipe, grid = _tab(_lr("l2")), {"clf__C": C_GRID}
    jobs = []
    for r, k, tr, te in outer_splits(y, g, N_SPLITS, N_REPEATS):
        for n_tr in (12, 18, 24, 30, 36):
            rng = np.random.default_rng(100 * r + 10 * k + n_tr)
            pos, neg = tr[y[tr] == 1], tr[y[tr] == 0]
            n_pos = max(3, int(round(n_tr * len(pos) / len(tr))))
            sub = np.r_[rng.choice(pos, min(n_pos, len(pos)), replace=False),
                        rng.choice(neg, min(n_tr - n_pos, len(neg)), replace=False)]
            for name in sets:
                jobs.append((name, r, k, n_tr, sub, te))

    def one(name, r, k, n_tr, sub, te):
        n_inner = int(min(N_INNER, y[sub].sum(), (1 - y[sub]).sum()))
        idx = np.r_[sub, te]
        _, prob, _ = _fit_predict(pipe, grid, sets[name][idx], y[idx], g[idx], np.arange(len(sub)),
                                  np.arange(len(sub), len(idx)), n_inner, 1000 * r + k, "roc_auc")
        return name, r, n_tr, g[te], y[te], prob

    res = Parallel(n_jobs=n_jobs)(delayed(one)(*j) for j in jobs)
    rows = []
    df = pd.DataFrame([(n, r, t, gi, yi, pi) for n, r, t, G, Y, P in res for gi, yi, pi in zip(G, Y, P)],
                      columns=["model", "repeat", "n_train", "subject_id", "y", "prob"])
    for (n, t, r), d in df.groupby(["model", "n_train", "repeat"]):
        rows.append({"model": n, "n_train": t, "repeat": r, "auc": subject_metrics(d["y"], d["prob"])["auc"]})
    lc = pd.DataFrame(rows).groupby(["model", "n_train"])["auc"].agg(["mean", "std"]).reset_index()
    lc.to_csv(out / "learning_curve.csv", index=False)
    print(lc.round(3).to_string())

14_rawrf_permutation.py: raw-amplitude RF permutation
"""Paso 14 (D6, exploratorio): test de permutación del RF de amplitudes crudas, separado por sujeto.

El control de inflación (``07_models.py --only control``) dio un AUC por sujeto de 0.68 / 0.75 / 0.66 en tres

- pliegues de ``outer_splits(seed0=0)``;
- RF con 300 árboles, ``max_features="sqrt"``, ``min_samples_leaf=3``, ``class_weight="balanced"`` y
  ``random_state=0``;
- 1 de cada 2 muestras de la señal regional.

Las etiquetas se permutan entre sujetos. p = (1 + #{AUC_perm ≥ AUC_obs}) / (1 + n_perm).
"""

import argparse

def run_once(Xf, y, g, n_jobs, seed=0):
    prob = np.zeros(len(y))
    for _, _, tr, te in outer_splits(y, g, 5, 1, seed0=seed):
        m = RandomForestClassifier(n_estimators=300, max_features="sqrt", min_samples_leaf=3,
                                   class_weight="balanced", n_jobs=n_jobs, random_state=seed).fit(Xf[tr], y[tr])
        prob[te] = m.predict_proba(Xf[te])[:, 1]
    s = pd.DataFrame({"g": g, "y": y, "p": prob}).groupby("g").agg(y=("y", "first"), p=("p", "mean"))
    return roc_auc_score(s["y"], s["p"])

    g = d["subject_id"].astype(str)

    null = []
    for i in range(args.n_perm):
        rng = np.random.default_rng(10_000 + i)
        mp = dict(zip(subj["g"], rng.permutation(subj["y"].to_numpy())))

    np.save(out / "rawrf_perm_null.npy", null)

         "seconds": round(time.time() - t0, 1)}

    print(r)

"""Fase C (D6, exploratorio): tamaño de muestra mínimo para detectar un AUC dado.

Cota optimista: una puntuación ya entrenada (sin costo de aprendizaje) con distribución binormal de varianzas
iguales, proporción MDD = 18/45 como en este estudio, prueba unilateral alfa = 0.05 con la nula exacta de
Mann-Whitney (aproximación normal) y semilla fija. Un pipeline que aprende de la misma muestra necesita más
(ver power_semisynthetic.csv). Salida: reports/results/audit/sample_size.csv.
"""
from pathlib import Path

import numpy as np
import pandas as pd
from scipy import stats

OUT = Path(__file__).resolve().parents[1] / "reports" / "results" / "audit"
P_MDD, ALPHA, SIMS = 18 / 45, 0.05, 4000
rng = np.random.default_rng(0)


def power(n, auc):
    n1 = max(2, round(n * P_MDD)); n0 = n - n1
    d = np.sqrt(2) * stats.norm.ppf(auc)
    thr = 0.5 + stats.norm.ppf(1 - ALPHA) * np.sqrt((n0 + n1 + 1) / (12 * n0 * n1))
    a = rng.normal(d, 1, (SIMS, n1)); b = rng.normal(0, 1, (SIMS, n0))
    r = stats.rankdata(np.concatenate([a, b], 1), axis=1)[:, :n1].sum(1)
    est = (r - n1 * (n1 + 1) / 2) / (n1 * n0)
    hw = 1.96 * np.std(est)
    return float((est >= thr).mean()), float(hw)


rows = []
for auc in (0.60, 0.65, 0.70, 0.75, 0.80):
    for n in (45, 60, 80, 100, 150, 200, 300, 400, 600, 800):
        p, hw = power(n, auc)
        rows.append({"true_auc": auc, "n_total": n, "power": p, "auc_ci_halfwidth": hw})
df = pd.DataFrame(rows)
need = (df[df.power >= 0.8].groupby("true_auc").n_total.min()
        .reindex(df.true_auc.unique()).rename("n_for_80pct_power"))
df.to_csv(OUT / "sample_size.csv", index=False)
need.reset_index().to_csv(OUT / "sample_size_needed.csv", index=False)
print(df.pivot(index="n_total", columns="true_auc", values="power").round(2).to_string())
print(need.to_string())

Copia al libro **solo resultados agregados**: figuras del artículo y tablas resumen por modelo. Ningún archivo con

from __future__ import annotations

import re
import shutil
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parents[1]
RES = ROOT / "reports" / "results"
OUT = ROOT / "book" / "_generated"

FIGURES = ["fig_model_comparison", "fig_roc", "fig_permutation", "fig_inflation", "fig_quality", "fig_regions",
           "fig_topomaps_power", "fig_erp_groups", "fig_family_importance", "fig_shap", "fig_power_curve", "fig_emg"]
ID_RE = re.compile(r"\b020[123]\d{4}\b")  # IDs de MODMA: no deben aparecer en tablas del libro


def ci(row, k, nd=2):
    return f"{row[k]:.{nd}f} [{row[k + '_lo']:.{nd}f}, {row[k + '_hi']:.{nd}f}]"


def fp(p):
    return "<0.001" if p < 0.001 else f"{p:.3f}"


def md_table(df: pd.DataFrame) -> str:
    head = "| " + " | ".join(df.columns) + " |\n|" + "---|" * len(df.columns) + "\n"
    body = "\n".join("| " + " | ".join(str(v) for v in r) + " |" for r in df.itertuples(index=False))
    text = head + body + "\n"
    assert not ID_RE.search(text), "una tabla del libro contiene un ID de participante"
    return text


def main() -> None:
    (OUT / "figures").mkdir(parents=True, exist_ok=True)
    missing = []

        src = RES / "figures" / f"{f}.png"

    print(f"libro: {len(FIGURES) - len(missing)} figuras, 3 tablas, desviaciones"
          + (f"; faltan {missing}" if missing else ""))


if __name__ == "__main__":
    main()
