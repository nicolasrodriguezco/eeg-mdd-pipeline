Evaluation protocol#

This is the core of the study. Every design choice serves one goal: the estimate must describe performance on a new person.

Element

Choice

Outer loop

Stratified 5-fold CV over participants, repeated 10 times (3 for deep learning)

Partitions

Identical for all models, so models can be compared pairwise

Inner loop

5-fold participant-grouped CV inside the training participants, maximising AUC

Leakage check

An assertion in every outer, inner and validation split

Unit of analysis

The participant. Trial-level probabilities are averaged per participant

Primary outcome

Participant-level ROC AUC, fixed in advance

Other metrics

Balanced accuracy, sensitivity and specificity at a fixed 0.5 threshold (never tuned), F1, Brier

Confidence intervals

2,000 stratified bootstrap resamples of participants on out-of-fold predictions

Significance

Labels permuted between participants and the full nested pipeline re-run (1,000 or 200 permutations)

Model comparison

Nadeau–Bengio corrected resampled \(t\)-test; Holm correction across secondary models

The permutation \(p\)-value is

\[ p = \frac{1 + \#\{\mathrm{AUC}_\text{perm} \ge \mathrm{AUC}_\text{obs}\}}{1 + n_\text{perm}}. \]

The leakage guard#

The assertion below runs in every split. If any participant appears on both sides, the run stops.

def assert_disjoint(groups: np.ndarray, train_idx: np.ndarray, test_idx: np.ndarray) -> None:
    overlap = set(np.asarray(groups)[train_idx]) & set(np.asarray(groups)[test_idx])
    if overlap:
        raise SubjectLeakError(f"Sujetos en entrenamiento y prueba a la vez: {sorted(overlap)[:5]}")

def outer_splits(y: np.ndarray, groups: np.ndarray, n_splits: int = 5, n_repeats: int = 10,
                 seed0: int = 0) -> Iterator[tuple[int, int, np.ndarray, np.ndarray]]:
    """(repetición, pliegue, train_idx, test_idx) sobre filas, con sujetos disjuntos."""
    groups = np.asarray(groups)
    for r in range(n_repeats):
        for k, (tr_s, te_s) in enumerate(subject_folds(y, groups, n_splits, seed0 + r)):
            tr = np.flatnonzero(np.isin(groups, list(tr_s)))
            te = np.flatnonzero(np.isin(groups, list(te_s)))
            assert_disjoint(groups, tr, te)
            yield r, k, tr, te

Nested cross-validation#

def nested_cv(name: str, estimator, param_grid, X, y, groups, n_splits: int = 5, n_repeats: int = 10,
              n_inner: int = 5, seed0: int = 0, n_jobs: int = 1, scoring: str = "roc_auc") -> CVResult:
    """CV anidada por sujeto. ``y`` y ``groups`` tienen una entrada por fila de ``X``
    (fila = sujeto o ensayo). Devuelve predicciones OOF por sujeto y métricas."""
    y = np.asarray(y).astype(int)
    groups = np.asarray(groups)
    jobs = list(outer_splits(y, groups, n_splits, n_repeats, seed0))
    outs = Parallel(n_jobs=n_jobs)(
        delayed(_fit_predict)(estimator, param_grid, X, y, groups, tr, te, n_inner, seed0 + 1000 * r + k, scoring)
        for r, k, tr, te in jobs)

    oof_rows, fold_rows, params = [], [], []
    for (r, k, _, _), (te, prob, best) in zip(jobs, outs):
        subj = aggregate_to_subject(prob, y[te], groups[te]).assign(repeat=r, fold=k)
        oof_rows.append(subj)
        fold_rows.append({"repeat": r, "fold": k, "n_test_subjects": len(subj),
                          **subject_metrics(subj["y"], subj["prob"])})
        params.append({"repeat": r, "fold": k, **{str(p): str(v) for p, v in best.items()}})
    oof = pd.concat(oof_rows, ignore_index=True)
    rep = pd.DataFrame([{"repeat": r, **subject_metrics(d["y"], d["prob"])} for r, d in oof.groupby("repeat")])
    return CVResult(name, oof, pd.DataFrame(fold_rows), rep, params)

def permutation_test(estimator, param_grid, X, y, groups, observed_auc: float, n_perm: int = 1000,
                     n_splits: int = 5, n_inner: int = 5, seed0: int = 0, n_jobs: int = 1) -> dict:
    """Test de permutación a nivel sujeto: se barajan las etiquetas ENTRE sujetos
    (cada ensayo hereda la etiqueta permutada de su sujeto) y se repite toda la
    CV anidada (1 repetición). p = (1 + #{AUC_perm >= AUC_obs}) / (1 + n_perm)."""
    y = np.asarray(y).astype(int)
    groups = np.asarray(groups)
    subj = pd.DataFrame({"g": groups, "y": y}).drop_duplicates("g")

    def one(i):
        rng = np.random.default_rng(10_000 + i)
        perm = dict(zip(subj["g"], rng.permutation(subj["y"].to_numpy())))
        yp = np.array([perm[g] for g in groups])
        res = nested_cv("perm", estimator, param_grid, X, yp, groups, n_splits, 1, n_inner, seed0, n_jobs=1)
        return float(res.repeat_scores["auc"].iloc[0])

    null = np.array(Parallel(n_jobs=n_jobs)(delayed(one)(i) for i in range(n_perm)))
    p = (1 + np.sum(null >= observed_auc)) / (1 + n_perm)
    return {"observed_auc": observed_auc, "n_perm": n_perm, "p_value": float(p),
            "null_mean": float(null.mean()), "null_q95": float(np.quantile(null, 0.95)), "null": null}

Participant-level metrics and bootstrap#

def bootstrap_ci(y: np.ndarray, prob: np.ndarray, n_boot: int = 2000, seed: int = 0,
                 alpha: float = 0.05) -> pd.DataFrame:
    """IC percentil por bootstrap estratificado sobre sujetos (una fila por sujeto)."""
    y = np.asarray(y).astype(int)
    prob = np.asarray(prob, dtype=float)
    rng = np.random.default_rng(seed)
    pos, neg = np.flatnonzero(y == 1), np.flatnonzero(y == 0)
    rows = []
    for _ in range(n_boot):
        idx = np.r_[rng.choice(pos, len(pos)), rng.choice(neg, len(neg))]
        rows.append(subject_metrics(y[idx], prob[idx]))
    boot = pd.DataFrame(rows)
    point = subject_metrics(y, prob)
    return pd.DataFrame({
        "estimate": pd.Series(point),
        "ci_low": boot.quantile(alpha / 2),
        "ci_high": boot.quantile(1 - alpha / 2),
    })
